Eglantine Launcher package repos

Fedora / Nobara (dnf)

sudo rpm --import https://savagecore.github.io/eglantine-launcher/dnf/RPM-GPG-KEY-eglantine-launcher
sudo curl -o /etc/yum.repos.d/eglantine-launcher.repo https://savagecore.github.io/eglantine-launcher/dnf/eglantine-launcher.repo
sudo dnf install eglantine-launcher

Debian / Ubuntu (apt)

echo "deb [trusted=yes] https://savagecore.github.io/eglantine-launcher/apt/ /" | sudo tee /etc/apt/sources.list.d/eglantine-launcher.list
sudo apt update && sudo apt install eglantine-launcher

Flatpak

flatpak remote-add --if-not-exists eglantine-launcher https://savagecore.github.io/eglantine-launcher/flatpak/eglantine-launcher.flatpakref
flatpak install eglantine-launcher dev.savagecore.EglantineLauncher
